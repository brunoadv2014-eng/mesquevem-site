-- schema.sql PRAGMA foreign_keys = ON; CREATE TABLE users ( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT NOT NULL, email TEXT UNIQUE NOT NULL, password_hash TEXT NOT NULL, role TEXT NOT NULL DEFAULT 'client', -- client | provider | admin created_at DATETIME DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE providers ( id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE, business_name TEXT, bio TEXT, location TEXT, price_currency TEXT DEFAULT 'BRL', base_reservation_price_cents INTEGER DEFAULT 5000, -- R$50.00 default created_at DATETIME DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE availability_rules ( id INTEGER PRIMARY KEY AUTOINCREMENT, provider_id INTEGER NOT NULL REFERENCES providers(id) ON DELETE CASCADE, weekday INTEGER, -- 0=domingo .. 6=sábado start_time TEXT, -- "09:00" end_time TEXT, -- "17:00" capacity INTEGER DEFAULT 1, created_at DATETIME DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE availability_exceptions ( id INTEGER PRIMARY KEY AUTOINCREMENT, provider_id INTEGER NOT NULL REFERENCES providers(id) ON DELETE CASCADE, date TEXT, -- "YYYY-MM-DD" is_available BOOLEAN DEFAULT 0, -- 0 = closed, 1 = open override note TEXT ); CREATE TABLE bookings ( id INTEGER PRIMARY KEY AUTOINCREMENT, provider_id INTEGER NOT NULL REFERENCES providers(id), client_id INTEGER NOT NULL REFERENCES users(id), date TEXT NOT NULL, -- "YYYY-MM-DD" time TEXT NOT NULL, -- "HH:MM" status TEXT NOT NULL DEFAULT 'reserved', -- reserved, quoted, accepted, completed, canceled, refunded created_at DATETIME DEFAULT CURRENT_TIMESTAMP, price_cents INTEGER NOT NULL, payment_intent_id TEXT, -- link to Stripe payment intent or placeholder payment_status TEXT DEFAULT 'pending' -- pending, paid, refunded ); CREATE TABLE reviews ( id INTEGER PRIMARY KEY AUTOINCREMENT, provider_id INTEGER NOT NULL REFERENCES providers(id), client_id INTEGER NOT NULL REFERENCES users(id), booking_id INTEGER REFERENCES bookings(id), rating INTEGER NOT NULL CHECK(rating BETWEEN 1 AND 5), comment TEXT, created_at DATETIME DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE provider_subscriptions ( id INTEGER PRIMARY KEY AUTOINCREMENT, provider_id INTEGER NOT NULL REFERENCES providers(id), stripe_subscription_id TEXT, status TEXT DEFAULT 'inactive', -- active | past_due | canceled | inactive started_at DATETIME, ended_at DATETIME );
